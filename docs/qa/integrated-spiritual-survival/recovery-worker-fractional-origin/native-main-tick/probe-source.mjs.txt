import assert from 'node:assert/strict';

import {Navigation,BIOME_IDS} from './src/world/navigation.js';

const bytes=readFileSync('tests/fixtures/worker-fractional-departure/state.json.gz');
function navigation(s){const profile=JSON.parse(readFileSync('public/content/biome-'+BIOME_IDS[s.biome]+'.json')).profile;const nav=new Navigation(s.seed,s.biome,profile);nav.setState(s);return nav;}
const rows=[];
for(const reload of [false,true]){
 let s=deserialize(gunzipSync(bytes).toString());
 // Explicit isolated locomotion fixture, not a resumed lost campaign.
 s.result=null;s.pauses=[];s.time=400;s.raid=null;s.spells=[];
 s.dayPlan={at:0,done:true,group:[]};s.nightPlan={at:500,done:true,group:[]};

 for(;steps<1200&&worker.status!=='home';steps++){
  const before={x:worker.x,z:worker.z};tick(s,.1,nav);maxStep=Math.max(maxStep,Math.hypot(worker.x-before.x,worker.z-before.z));
  if(reload&&steps===100){s=deserialize(serialize(s));worker=s.workers[0];nav=navigation(s);reloads++;}
 }

 rows.push({reload,steps,reloads,start,end:{x:worker.x,z:worker.z},status:worker.status,maxStep,navigationVersion:nav.version,ledgerUnchanged:true,clockAdvanced:s.time-400});
}
const result={observedAt:new Date().toISOString(),originalSnapshotSha256:createHash('sha256').update(bytes).digest('hex'),scope:'Synthetic isolated locomotion fixture derived from archived geometry; actual native tick and persistence.
